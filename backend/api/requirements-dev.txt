# Test-only pins. Kept beside the api's runtime requirements (backend/api/requirements.txt)
# because the tests drive the api, so their versions belong with it — but nothing here
# goes into any built image.
#
# httpx is already a runtime dependency (pinned there); pytest is the runner for
# tests/. Both pinned to stable majors so a new x release cannot silently change
# what a failing test means.
pytest==8.4.2
httpx==0.28.1
# The api's own runtime, because several unit suites import api modules
# directly (tests/zero-downtime/test_storage_retryable.py loads
# services.storage, which needs fastapi and boto3). Without it pytest stops at
# collection and not one test runs — which is what CI did until this line.
-r requirements.txt
# icalendar parses the RFC 5545 files the .ics tests assert against — a wire
# format, which eyeballing would not test at all. Pinned because the tests
# importorskip it, so without a pin a missing install turns every parse
# assertion into a silent skip instead of a failure.
icalendar==6.3.1
# The passkey suite's software authenticator (tests/passkeys/) makes real P-256
# keys and CBOR-encodes what a phone would send, so the api's WebAuthn checks
# are tested against real cryptography. Same libraries py_webauthn uses.
cryptography>=44,<47
cbor2==6.1.4
